import org.apache.spark.sql.functions._
import spark.implicits._

val lines = spark.readStream.format("socket").option("host","localhost").option("port",9999).load()

val textData = lines.as[String]

val stopWords = Seq("is","the","a","an","and","to")

val cleaned = textData.map(line => { val lower = line.toLowerCase; val noSpace = lower.trim.replaceAll("\\s+"," "); val noPunct = noSpace.replaceAll("[^a-zA-Z ]",""); val words = noPunct.split(" ").filter(w => w.nonEmpty && !stopWords.contains(w)); val lemmatized = words.map{ case w if w.endsWith("ing") && w.length > 4 => w.dropRight(3); case w if w.endsWith("ed") && w.length > 3 => w.dropRight(2); case w => w }; lemmatized.mkString(" ") })

val query = cleaned.writeStream.outputMode("append").format("console").option("truncate","false").start()

query.awaitTermination()
